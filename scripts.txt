1///////////
MATCH (e:Equipamento)
OPTIONAL MATCH (e)-[:AFETADO_POR]->(v:Vulnerabilidade)
WITH e, coalesce(SUM(v.cvss), 0.0) * e.fator_exposicao AS b_i
SET e.risco_proprio = b_i
RETURN e.nome AS Ativo, e.risco_proprio AS RiscoProprio;

2///////////////
// 1. Criar a projeção em memória
CALL gds.graph.project(
  'grafoRisco',
  'Equipamento',
  {
    DEPENDE_DE: {
      type: 'DEPENDE_DE',
      orientation: 'NATURAL',
      properties: 'peso'
    }
  },
  {
    nodeProperties: ['risco_proprio']
  }
);

// 2. Executar o PageRank Ponderado e gravar o risco efetivo direto nos nós
CALL gds.pageRank.write('grafoRisco', {
  writeProperty: 'risco_efetivo',
  relationshipWeightProperty: 'peso',
  sourceNodes: [],
  dampingFactor: 0.85,
  maxIterations: 20
})
YIELD nodePropertiesWritten, computeMillis;

// 3. Remover a projeção da memória
CALL gds.graph.drop('grafoRisco');

3/////////////////////
MATCH (e:Equipamento)
OPTIONAL MATCH (e)-[r:DEPENDE_DE]->(dep:Equipamento)
OPTIONAL MATCH (e)-[vuln_rel:AFETADO_POR]->(v:Vulnerabilidade)
RETURN e, r, dep, vuln_rel, v

4///////
// 1. Calcula a soma total de risco próprio no sistema
MATCH (e:Equipamento)
WITH sum(e.risco_proprio) AS risco_total

// 2. Ajusta o risco efetivo para a escala CVSS
MATCH (e:Equipamento)
SET e.risco_efetivo_escalonado = e.risco_efetivo * risco_total
RETURN e.nome, e.risco_proprio, e.risco_efetivo, e.risco_efetivo_escalonado;

5////////////////
CALL gds.pageRank.write('grafoRisco', {
  writeProperty: 'risco_efetivo',
  relationshipWeightProperty: 'peso',
  sourceNodes: [],
  dampingFactor: 0.85,
  maxIterations: 20
})

6/////////////////
// 1. Calcula a soma do Risco Próprio
MATCH (e:Equipamento)
WITH sum(e.risco_proprio) AS soma_risco_total

// 2. Desnormaliza o Risco Efetivo do GDS para a escala do CVSS
MATCH (e:Equipamento)
SET e.risco_efetivo_cvss = e.risco_efetivo * soma_risco_total
RETURN e.nome AS Ativo,
       e.risco_proprio AS RiscoProprio,
       e.risco_efetivo AS ProporcaoGDS,
       e.risco_efetivo_cvss AS RiscoEfetivoCVSS
ORDER BY RiscoEfetivoCVSS DESC;

CREATE (app_server)-[:DEPENDE_DE {peso: 0.20}]->(worker)
